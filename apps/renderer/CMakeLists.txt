cmake_minimum_required(VERSION 3.26)
project(ShowNightRenderer VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Der erste native Renderer unterstützt ausschließlich Windows x64.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
set(SKIA_ROOT "" CACHE PATH "Festgelegter Skia-Quellstand")
set(GSTREAMER_ROOT "" CACHE PATH "Offizielles GStreamer-MSVC-SDK")
set(JSON_INCLUDE "" CACHE PATH "Festgelegter nlohmann-JSON-Header")
set(SKIA_REVISION "" CACHE STRING "SHA des gebauten Skia-Quellstands")
if(NOT EXISTS "${SKIA_ROOT}/out/ShowNight/skia.lib")
  message(FATAL_ERROR "Geprüfter Skia-Build fehlt.")
endif()
add_executable(shownight-renderer src/main.cpp)
target_compile_definitions(shownight-renderer PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN SKIA_REVISION="${SKIA_REVISION}")
target_compile_options(shownight-renderer PRIVATE /W4 /EHsc /utf-8)
target_include_directories(shownight-renderer PRIVATE "${SKIA_ROOT}" "${JSON_INCLUDE}" "${GSTREAMER_ROOT}/include/gstreamer-1.0" "${GSTREAMER_ROOT}/include/glib-2.0" "${GSTREAMER_ROOT}/lib/glib-2.0/include")
target_link_directories(shownight-renderer PRIVATE "${GSTREAMER_ROOT}/lib")
target_link_libraries(shownight-renderer PRIVATE "${SKIA_ROOT}/out/ShowNight/skia.lib" "${SKIA_ROOT}/out/ShowNight/libpng.lib" "${SKIA_ROOT}/out/ShowNight/zlib.lib" gstreamer-1.0 gstapp-1.0 gstvideo-1.0 gobject-2.0 glib-2.0 advapi32 dwrite gdi32 ole32 user32 usp10)
set_target_properties(shownight-renderer PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
